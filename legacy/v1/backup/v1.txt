def resource_path(relative_path: str) -> str:
    """Get absolute path to resource, works for development and PyInstaller."""
    try:
        # PyInstaller creates a temp folder and stores path in _MEIPASS
        base_path = sys._MEIPASS
    except Exception:
        # If not running as a bundled app, use the script's directory
        base_path = os.path.abspath(os.path.join(os.path.dirname(__file__), ".")) # More robust way to get base path

    return os.path.join(base_path, relative_path)

    elif sys.platform == "darwin":

PROJECTS_FILE = os.path.join(config_dir, "projects.json")
AUTO_LAUNCH_FILE = os.path.join(config_dir, "auto_launch.json")
SETTINGS_FILE = os.path.join(config_dir, "settings.json") # For storing dark mode preference


def load_settings():
    if os.path.exists(SETTINGS_FILE):
        try:
            with open(SETTINGS_FILE, "r") as f:
                return json.load(f)
        except json.JSONDecodeError:
            return {} # Return empty dict if corrupted
    return {}

def save_settings(settings):
    with open(SETTINGS_FILE, "w") as f:
        json.dump(settings, f, indent=4)

def load_projects():
    if os.path.exists(PROJECTS_FILE):
        try:
            with open(PROJECTS_FILE, "r") as f:
                projects = json.load(f)
                # Convert file entries from plain string to dict if needed.
                for proj in projects.values():
                    new_files = []
                    for item in proj.get("files", []):
                        if isinstance(item, dict) and "path" in item and "selected" in item:
                            new_files.append(item)
                        elif isinstance(item, str): # Handle old format
                            new_files.append({"path": item, "selected": True})
                        # Skip malformed entries silently or log a warning
                    proj["files"] = new_files
                return projects
        except json.JSONDecodeError:
            # Use a non-blocking way to show error if possible, or log it
            print(f"Error: Projects file '{PROJECTS_FILE}' corrupted. Creating new file.", file=sys.stderr)
            # QMessageBox.critical(None, "Error", "Projects file corrupted. Creating new file.") # Avoid blocking on startup
            return {}
    return {}

def save_projects(projects):
    with open(PROJECTS_FILE, "w") as f:
        json.dump(projects, f, indent=4)

def load_auto_launch():
    if os.path.exists(AUTO_LAUNCH_FILE):
        try:
            with open(AUTO_LAUNCH_FILE, "r") as f:
                return json.load(f)
        except json.JSONDecodeError:
            print(f"Error: Auto-launch file '{AUTO_LAUNCH_FILE}' corrupted. Creating new file.", file=sys.stderr)
            return {}
    return {}

def save_auto_launch(auto_launch):

# Helper function to adjust color brightness for hover effect
def adjust_color_brightness(hex_color, factor=1.2):
    """ Increases brightness for hover effect. Factor > 1 lightens, < 1 darkens. """
    try:
        color = QColor(hex_color)
        # Ensure factor doesn't blow out colors or make them black
        h, s, v, a = color.getHsv()
        new_v = min(255, int(v * factor))
        hover_color = QColor.fromHsv(h, s, new_v, a)
        return hover_color.name()
    except Exception:
        return hex_color # Return original if conversion fails

# Custom File List Widget (Delete key support)
# ---------------------
class FileListWidget(QListWidget):
    def __init__(self, parent=None): # Removed file_list argument, not needed directly
        super().__init__(parent)
        self.setSelectionMode(QListWidget.SelectionMode.ExtendedSelection) # Allow multi-select for deletion

    def keyPressEvent(self, event: QKeyEvent):
        if event.key() == Qt.Key.Key_Delete:
            selected_items = self.selectedItems()
            if selected_items:
                # Ask for confirmation if deleting multiple items? (Optional)
                reply = QMessageBox.question(self, "Confirm Deletion",
                                             f"Remove {len(selected_items)} selected item(s)?",
                                             QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
                                             QMessageBox.StandardButton.No)
                if reply == QMessageBox.StandardButton.Yes:
                    for item in selected_items:
                        row = self.row(item)
                        self.takeItem(row)
                    # Note: The actual data deletion needs to happen in the dialog's save logic
        else:
            super().keyPressEvent(event)

    def apply_style(self):
        qss = generate_qss(self.dark_mode)
        self.setStyleSheet(qss)
        # Re-apply random colors after main style changes
        self.apply_random_card_colors()
        self.statusBar().showMessage(f"{'Dark' if self.dark_mode else 'Light'} mode enabled. Ready.")

    def apply_random_card_colors(self):
        """Iterates through existing cards and applies random colors."""
        palette = CARD_COLOR_PALETTE_DARK if self.dark_mode else CARD_COLOR_PALETTE_LIGHT
        if not palette: return # No colors defined

        for i in range(self.projects_layout.count()):
            widget = self.projects_layout.itemAt(i).widget()
            if isinstance(widget, QFrame) and widget.objectName() == "ProjectCard":
                base_color = random.choice(palette)
                hover_color = adjust_color_brightness(base_color, 1.15 if self.dark_mode else 0.9) # Darken slightly on light mode hover

                # Apply specific background and hover effect using stylesheet
                widget.setStyleSheet(f"""
                    QFrame#ProjectCard {{
                        background-color: {base_color};
                        border: 1px solid {adjust_color_brightness(base_color, 1.2)}; /* Border slightly lighter/darker */
                        border-radius: 10px;
                    }}
                    QFrame#ProjectCard:hover {{
                        background-color: {hover_color};
                    }}
                    /* Reset nested widget styles that might be affected */
                    QFrame#ProjectCard QLabel, QFrame#ProjectCard QPushButton, QFrame#ProjectCard QCheckBox {{
                        background-color: transparent;
                    }}
                """)

    def toggle_dark_mode(self, checked):
        self.dark_mode = checked
        self.settings["dark_mode"] = self.dark_mode # Save preference
        save_settings(self.settings)
        self.apply_style() # Regenerate and apply QSS

    def refresh_projects(self):
        # Clear existing widgets more safely
        while self.projects_layout.count():
            item = self.projects_layout.takeAt(0)
            widget = item.widget()
            if widget is not None:
                widget.deleteLater() # Use deleteLater for safety

        if not self.projects:
            no_proj_label = QLabel("No projects found. Click 'Add Project' to create one.")
            no_proj_label.setAlignment(Qt.AlignmentFlag.AlignCenter)
            self.projects_layout.addWidget(no_proj_label)
        else:
            # Add project cards
            for name in sorted(self.projects.keys()):
                 # Check if project data is valid before creating card
                if isinstance(self.projects.get(name), dict):
                    self.projects_layout.addWidget(self.create_project_card(name))
                else:
                    print(f"Warning: Skipping invalid project data for '{name}'", file=sys.stderr)

            # Add a stretch at the end to push cards up if there are few
            # self.projects_layout.addStretch(1) # Add stretch *after* cards

        # Apply random colors *after* all cards are potentially added/removed
        self.apply_random_card_colors()
        # Ensure the layout updates visually
        self.projects_container.updateGeometry()
        self.scroll_area.updateGeometry()

    def add_project(self):

    def open_project(self, name):
        if name not in self.projects:
            QMessageBox.critical(self, "Error", f"Project '{name}' not found.")
            return

        project_data = self.projects[name]
        files_to_open = project_data.get("files", [])
        opened_any = False
        missing_files = []

        if not files_to_open:
             QMessageBox.information(self, "Empty Project", f"Project '{name}' has no files or links.")
             self.statusBar().showMessage(f"Project '{name}' is empty")
             return

        for item in files_to_open:
            # Ensure item is a dictionary with path and selected status
            if not isinstance(item, dict) or "path" not in item:
                print(f"Warning: Skipping malformed item in project '{name}': {item}", file=sys.stderr)
                continue

            if not item.get("selected", True): # Skip if not selected
                continue

            path = item["path"]
            if not path: # Skip empty paths
                continue

            try:
                is_url = path.lower().startswith("http://") or path.lower().startswith("https://")
                if is_url:
                    QDesktopServices.openUrl(QUrl(path))
                    opened_any = True
                else:
                    # Check if local file or directory exists
                    if os.path.exists(path):
                        QDesktopServices.openUrl(QUrl.fromLocalFile(os.path.abspath(path))) # Use absolute path
                        opened_any = True
                    else:
                        missing_files.append(path)
            except Exception as e:
                print(f"Error opening '{path}': {e}", file=sys.stderr)
                QMessageBox.warning(self, "Opening Error", f"Could not open:\n{path}\n\nError: {e}")

        if missing_files:
            QMessageBox.warning(self, "Missing Files/Folders",
                                f"The following items in project '{name}' could not be found:\n\n" + "\n".join(missing_files))

        if opened_any:
            self.statusBar().showMessage(f"Attempted to open selected items in '{name}'")
        elif not missing_files: # No items were selected or valid
             self.statusBar().showMessage(f"No selected items to open in project '{name}'")


    def edit_project(self, name):
        if name not in self.projects:
            return

        dialog = EditProjectDialog(name, self.projects[name], self.dark_mode, self) # Pass dark mode status
        if dialog.exec() == QDialog.DialogCode.Accepted:
            new_name = dialog.project_name # Get potentially changed name
            new_data = {"files": dialog.files, "notes": dialog.notes}

            # Handle potential name change
            if new_name != name:
                 # Check if new name conflicts
                if new_name in self.projects:
                    QMessageBox.critical(self, "Error", f"Project name '{new_name}' already exists!")
                    # Re-open dialog? Or just fail? Let's fail for simplicity.
                    return
                # Update projects dictionary
                del self.projects[name]
                self.projects[new_name] = new_data
                # Update auto-launch dictionary
                if name in self.auto_launch:
                    self.auto_launch[new_name] = self.auto_launch.pop(name) # More efficient way to rename key
            else:
                # Just update the data if name is the same
                 self.projects[name] = new_data

            save_projects(self.projects)
            save_auto_launch(self.auto_launch) # Save auto-launch in case keys changed
            self.statusBar().showMessage(f"Project '{new_name}' updated")
            self.refresh_projects()

    def remove_project(self, name):
        reply = QMessageBox.question(self, "Confirm Removal", f"Are you sure you want to remove project '{name}'?\nThis cannot be undone.",
                                     QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.Cancel, # Use Cancel
                                     QMessageBox.StandardButton.Cancel) # Default to Cancel
        if reply == QMessageBox.StandardButton.Yes:
            if name in self.projects:
                del self.projects[name]
            if name in self.auto_launch:
                del self.auto_launch[name]
                save_auto_launch(self.auto_launch) # Save change
            save_projects(self.projects)
            self.statusBar().showMessage(f"Project '{name}' removed")
            self.refresh_projects()

    def toggle_auto_launch(self, name, state):
        # The state passed from the signal is the CheckState enum value (0, 1, 2)
        # We need boolean True if state is Checked (value 2)
        is_checked = bool(state) # Convert state (int) to bool
        self.auto_launch[name] = is_checked
        status_msg = f"Auto-launch {'enabled' if is_checked else 'disabled'} for '{name}'"
        self.statusBar().showMessage(status_msg)
        save_auto_launch(self.auto_launch)
        # No need to refresh projects visually for this change, just update the data file

    def import_projects(self):
        file_path, _ = QFileDialog.getOpenFileName(self, "Import Projects", "",
                                                   "JSON Files (*.json);;All Files (*)")
        if file_path:
            try:
                with open(file_path, "r") as f:
                    import_data = json.load(f)

                imported_projects = import_data.get("projects", {})
                imported_auto_launch = import_data.get("auto_launch", {})
                merge = False

                if not imported_projects and not imported_auto_launch:
                     QMessageBox.warning(self, "Import Error", "The selected file does not contain valid 'projects' or 'auto_launch' data.")
                     return

                if imported_projects:
                    reply = QMessageBox.question(self, "Import Projects",
                                                 "Choose import mode:\n"
                                                 "- **Merge:** Add new projects and update existing ones.\n"
                                                 "- **Replace:** Delete all current projects before importing.",
                                                 QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No | QMessageBox.StandardButton.Cancel,
                                                 QMessageBox.StandardButton.Cancel) # Buttons map to Merge, Replace, Cancel

                    if reply == QMessageBox.StandardButton.Cancel:
                        self.statusBar().showMessage("Import cancelled")
                        return
                    elif reply == QMessageBox.StandardButton.No: # Replace
                        self.projects = {} # Clear existing projects
                        self.auto_launch = {} # Clear existing auto-launch as well
                        merge = False
                    else: # Merge (Yes button)
                        merge = True

                    # Perform project import based on choice
                    if merge:
                        project_update_count = 0
                        project_add_count = 0
                        for proj_name, data in imported_projects.items():
                            # Basic validation of imported data structure
                            if isinstance(data, dict) and isinstance(data.get("files"), list):
                                if proj_name in self.projects:
                                    project_update_count += 1
                                else:
                                    project_add_count += 1
                                self.projects[proj_name] = data # Add or overwrite
                            else:
                                print(f"Warning: Skipping invalid project structure for '{proj_name}' during merge.", file=sys.stderr)
                        import_proj_msg = f"Merged projects: {project_add_count} added, {project_update_count} updated."
                    else: # Replace
                        self.projects = imported_projects
                        import_proj_msg = f"Replaced projects with {len(self.projects)} imported items."

                else:
                    import_proj_msg = "No 'projects' data found in the file."


                # Import auto-launch settings (always merge/update)
                auto_launch_update_count = 0
                if imported_auto_launch:
                    for proj_name, value in imported_auto_launch.items():
                         # Only apply if the project exists after project import
                         if proj_name in self.projects and isinstance(value, bool):
                            self.auto_launch[proj_name] = value
                            auto_launch_update_count +=1
                    import_auto_msg = f"Updated {auto_launch_update_count} auto-launch settings."
                else:
                    import_auto_msg = "No 'auto_launch' data found."


                # Save and refresh
                save_projects(self.projects)
                save_auto_launch(self.auto_launch)
                self.refresh_projects() # Refresh UI fully
                self.statusBar().showMessage("Import completed.")
                QMessageBox.information(self, "Import Summary", f"{import_proj_msg}\n{import_auto_msg}")

            except json.JSONDecodeError:
                 QMessageBox.critical(self, "Import Error", "Failed to read JSON file. It might be corrupted.")
                 print(f"Import Error: JSONDecodeError in {file_path}", file=sys.stderr)
            except Exception as e:
                QMessageBox.critical(self, "Import Error", f"An unexpected error occurred during import: {str(e)}")
                print(f"Import Error: {e}", file=sys.stderr)

    def auto_launch_projects(self):
        auto_launched = []
        for project_name, enabled in self.auto_launch.items():
            if enabled and project_name in self.projects:
                 # Check if project has *selected* files before attempting to open
                 has_selected_files = any(item.get("selected", True) for item in self.projects[project_name].get("files", []) if isinstance(item, dict))
                 if has_selected_files:
                     print(f"Auto-launching: {project_name}") # Log auto-launch attempt
                     self.open_project(project_name)
                     auto_launched.append(project_name)
        if auto_launched:
            # Show status briefly
            QTimer.singleShot(1000, lambda: self.statusBar().showMessage(f"Auto-launched: {', '.join(auto_launched)}", 5000)) # Show for 5 secs
        else:
            # If nothing launched, keep "Ready" or previous message
            pass

    def populate_files_list(self):
        self.files_list.clear()
        for file_item in self.files:
            path = file_item.get("path", "")
            selected = file_item.get("selected", True)
            item = QListWidgetItem(path)
            item.setFlags(item.flags() | Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsEnabled)
            item.setCheckState(Qt.CheckState.Checked if selected else Qt.CheckState.Unchecked)
            self.files_list.addItem(item)

    def add_file_item(self, path, selected=True):
        """Adds a new item to the list widget and internal list, avoiding duplicates."""
        # Check if path already exists
        for i in range(self.files_list.count()):
            if self.files_list.item(i).text() == path:
                 QMessageBox.information(self, "Duplicate Item", f"'{os.path.basename(path)}' is already in the list.")
                 return

        # Add to internal list (for saving later)
        # self.files.append({"path": path, "selected": selected}) # This will be rebuilt on save

        # Add to QListWidget
        item = QListWidgetItem(path)
        item.setFlags(item.flags() | Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsEnabled)
        item.setCheckState(Qt.CheckState.Checked if selected else Qt.CheckState.Unchecked)
        self.files_list.addItem(item)
        self.files_list.scrollToItem(item) # Ensure visible


    def add_file(self):
        # Allow selecting multiple files
        file_paths, _ = QFileDialog.getOpenFileNames(self, "Select File(s)")
        if file_paths:
            for file_path in file_paths:
                 self.add_file_item(os.path.normpath(file_path)) # Normalize path

    def add_folder(self):
        folder_path = QFileDialog.getExistingDirectory(self, "Select Folder")
        if folder_path:
            self.add_file_item(os.path.normpath(folder_path)) # Normalize path

    def add_link(self):
        link, ok = QInputDialog.getText(self, "Add Web Link", "Enter URL (e.g., https://example.com):")
        if ok and link.strip():
            clean_link = link.strip()
            if not (clean_link.lower().startswith("http://") or clean_link.lower().startswith("https://")):
                # Optionally prepend http:// if missing, or warn user
                reply = QMessageBox.question(self, "Add http://?", f"The link doesn't start with http:// or https://. Add 'http://' prefix?",
                                             QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No | QMessageBox.StandardButton.Cancel)
                if reply == QMessageBox.StandardButton.Yes:
                    clean_link = "http://" + clean_link
                elif reply == QMessageBox.StandardButton.Cancel:
                    return # User cancelled
                # If No, add the link as is
            self.add_file_item(clean_link)

    # --- Drag and Drop ---
    def dragEnterEvent(self, event):
        if event.mimeData().hasUrls():
            # Maybe change background slightly to indicate drop target?
            event.acceptProposedAction()
        else:
            event.ignore()

    def dragMoveEvent(self, event):
         if event.mimeData().hasUrls():
            event.acceptProposedAction()
         else:
            event.ignore()

    def dropEvent(self, event):
        if event.mimeData().hasUrls():
            event.setDropAction(Qt.DropAction.CopyAction) # Indicate copying
            event.acceptProposedAction()
            urls = event.mimeData().urls()
            for url in urls:
                if url.isLocalFile():
                    local_path = os.path.normpath(url.toLocalFile())
                    self.add_file_item(local_path)
                else:
                    # Assume it's a web link if not local
                    self.add_file_item(url.toString())
        else:
            event.ignore()

    # --- Save Logic ---
    def save_changes(self):
        new_name = self.name_edit.text().strip()
        if not new_name:
            QMessageBox.warning(self, "Invalid Name", "Project name cannot be empty.")
            return # Prevent saving with empty name

        # If name changed, check for duplicates (handled in MainWindow after dialog accepts)
        # We just pass the new name back.

        self.project_name = new_name # Update internal name
        self.notes = self.notes_edit.toPlainText().strip()

        # Rebuild the files list from the QListWidget state
        new_files_list = []
        for index in range(self.files_list.count()):
            item = self.files_list.item(index)
            path = item.text()
            is_selected = item.checkState() == Qt.CheckState.Checked
            if path: # Ensure path is not empty
                new_files_list.append({"path": path, "selected": is_selected})
        self.files = new_files_list

        self.accept() # Close dialog successfully
